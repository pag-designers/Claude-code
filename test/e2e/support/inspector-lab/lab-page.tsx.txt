'use client';

import * as React from 'react';
import { Inspector } from '@/components/editor/inspector';
import * as core from '@/lib/editor/core';
import { useEditorStore } from '@/lib/editor/core';
import { buildLabProject } from './lab-project';

/**
 * TEST-ONLY page (not part of the app): mounts the Inspector beside a dark stage with the store exposed as
 * window.__editor so Playwright can load a project, select items and read the resulting state.
 */
export default function InspectorLabPage() {
  React.useEffect(() => {
    const w = window as unknown as { __editor?: unknown; __lab?: unknown };
    w.__editor = useEditorStore;
    w.__lab = { buildLabProject };
    const q = new URLSearchParams(window.location.search);
    if (q.get('autoload') !== '0') {
      const st = useEditorStore.getState();
      st.load({ id: 'lab-project', name: 'Inspector lab', rev: 1, project: buildLabProject(), updatedAt: 0 });
    }
    // the editor's global keyboard handler (real core only): keys the inspector handles must not also reach it
    const handleKey = (core as unknown as { handleKey?: (e: KeyboardEvent, s: unknown) => boolean }).handleKey;
    if (handleKey) window.addEventListener('keydown', (e) => handleKey(e, useEditorStore));
    // record every store action call (works with the real store and the fake) so tests can count what the inspector asked for
    const log: { fn: string; args: unknown[] }[] = [];
    const wrapped: Record<string, unknown> = {};
    for (const [k, v] of Object.entries(useEditorStore.getState() as unknown as Record<string, unknown>)) {
      if (typeof v !== 'function') continue;
      wrapped[k] = (...args: unknown[]) => {
        log.push({ fn: k, args });
        return (v as (...a: unknown[]) => unknown)(...args);
      };
    }
    useEditorStore.setState(wrapped as never);
    (window as unknown as { __editorLog?: unknown }).__editorLog = log;
    (window as unknown as { __labReady?: boolean }).__labReady = true;
  }, []);
  return (
    <div className="fixed inset-0 z-50 flex bg-background" data-testid="inspector-lab">
      <div className="flex min-w-0 flex-1 items-center justify-center text-sm text-subtle">
        Inspector lab
      </div>
      <div className="h-full w-[340px] shrink-0 border-l border-border">
        <Inspector />
      </div>
    </div>
  );
}
